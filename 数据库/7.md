# 应用

## 核心 API 函数说明
1. `mysql_init()`
作用：初始化 MYSQL 句柄，创建空连接对象
2. `mysql_real_connect()`
作用：建立 TCP 连接到 MySQL 服务
3. `mysql_query(MYSQL*, const char* sql)`
作用：执行普通 SQL（增删改查）
4. `mysql_store_result(MYSQL*)`
作用：一次性把查询全部结果读到本地内存
5. `mysql_fetch_row(MYSQL_RES*)`
作用：循环读取一行数据
6. `mysql_affected_rows(MYSQL*)`
作用：获取 insert/update/delete 影响的数据行数
7. `mysql_error(MYSQL*)`
作用：获取当前连接最近一次操作的错误字符串
8. `mysql_free_result(MYSQL_RES*)`
作用：释放查询结果集占用内存
9. `mysql_close(MYSQL*)`
作用：关闭数据库 TCP 连接、释放句柄资源
10. `mysql_set_character_set(MYSQL*, const char* charset)`
设置连接字符集，代码用 utf8mb4
11. `mysql_num_fields(MYSQL_RES*)`
获取查询结果的列数量
12. `mysql_ping(MYSQL*)`
检测连接是否存活，断开则自动重连
13. `mysql_stmt_init(MYSQL*)`
创建预处理语句句柄 MYSQL_STMT
14. `mysql_stmt_prepare(MYSQL_STMT*, const char* sql, unsigned long len)`
预编译带？占位符的 SQL
15. `mysql_stmt_bind_param(MYSQL_STMT*, MYSQL_BIND*)`
绑定 SQL 占位符对应的参数
16. `mysql_stmt_execute(MYSQL_STMT*)`
执行预处理语句
17. `mysql_stmt_error(MYSQL_STMT*)`
获取预处理语句报错信息
18. `mysql_stmt_close(MYSQL_STMT*)`
销毁预处理句柄，释放资源

## MySQL 原生结构体 / 类型（API 配套）
1. MYSQL：数据库连接句柄主体
2. MYSQL_RES：查询结果集结构体
3. MYSQL_ROW：单行数据（char** 字符串数组）
4. MYSQL_STMT：预处理语句句柄
5. MYSQL_BIND：预处理参数绑定结构体
6. MYSQL_TYPE_STRING：绑定参数类型标记（字符串）

```cpp
DBHelper.h
#ifndef CONN_POOL_H
#define CONN_POOL_H

#include <mysql/mysql.h>
#include <string>
#include <vector>
#include <queue>
#include <mutex>
#include <condition_variable>
#include <thread>
#include <atomic>
#include <chrono>
#include <iostream>
#include <cstring>

// 单行/结果集定义
using RowData = std::vector<std::string>;   //一行数据库数据（比如 name、age、id）
using ResultSet = std::vector<RowData>;     //整张查询结果（很多行数据）

// 单个数据库连接封装
struct DbConn
{
    MYSQL mysql;
    std::chrono::steady_clock::time_point lastUse; // 上次使用时间
    DbConn()
    {
        mysql_init(&mysql);
        lastUse = std::chrono::steady_clock::now();
    }
    ~DbConn()
    {
        mysql_close(&mysql);
    }
    // 执行查询
    ResultSet Query(const std::string& sql)
    {
        ResultSet resSet;
        if (mysql_query(&mysql, sql.c_str()))
        {
            std::cout << "SQL err:" << mysql_error(&mysql) << " sql:" << sql << std::endl;
            return resSet;
        }
        MYSQL_RES* res = mysql_store_result(&mysql);
        if (!res) return resSet;
        int col = mysql_num_fields(res);
        MYSQL_ROW row;
        while ((row = mysql_fetch_row(res)))
        {
            RowData line;
            for (int i = 0; i < col; i++)
                line.push_back(row[i] ? row[i] : "");
            resSet.push_back(line);
        }
        mysql_free_result(res);
        lastUse = std::chrono::steady_clock::now();
        return resSet;
    }
    // 增删改
    int Update(const std::string& sql)
    {
        if (mysql_query(&mysql, sql.c_str()))
        {
            std::cout << "SQL err:" << mysql_error(&mysql) << " sql:" << sql << std::endl;
            return -1;
        }
        lastUse = std::chrono::steady_clock::now();
        return static_cast<int>(mysql_affected_rows(&mysql));
    }
    // 预处理防注入（仅string参数）
    bool PrepareExec(const std::string& sql, const std::vector<std::string>& params)
    {
        MYSQL_STMT* stmt = mysql_stmt_init(&mysql);
        if (!stmt) return false;
        if (mysql_stmt_prepare(stmt, sql.c_str(), sql.size()))
        {
            std::cout << mysql_stmt_error(stmt) << std::endl;
            mysql_stmt_close(stmt);
            return false;
        }
        int cnt = params.size();
        std::vector<MYSQL_BIND> bind(cnt);
        std::vector<unsigned long> len(cnt);
        for (int i = 0; i < cnt; i++)
        {
            memset(&bind[i], 0, sizeof(MYSQL_BIND));
            bind[i].buffer_type = MYSQL_TYPE_STRING;
            bind[i].buffer = const_cast<char*>(params[i].data());
            len[i] = params[i].size();
            bind[i].buffer_length = len[i];
            bind[i].length = &len[i];
        }
        mysql_stmt_bind_param(stmt, bind.data());
        bool ret = !mysql_stmt_execute(stmt);
        mysql_stmt_close(stmt);
        lastUse = std::chrono::steady_clock::now();
        return ret;
    }
    // 重置连接状态（归还前清理）
    void Reset()
    {
        mysql_ping(&mysql); // 检测连接是否断开，自动重连
    }
};

// 连接池单例
class MysqlConnPool
{
public:
    static MysqlConnPool& GetInstance();
    // 初始化连接池参数
    bool Init(const std::string& host, const std::string& user, const std::string& pwd,
              const std::string& db, int port, int minConn, int maxConn,
              int idleTimeout = 60, int getConnTimeout = 3);
    // 获取连接，超时返回nullptr
    DbConn* GetConn();
    // 归还连接
    void RetConn(DbConn* conn);
    // 关闭池，释放所有连接
    void ClosePool();

private:
    MysqlConnPool();
    ~MysqlConnPool();
    MysqlConnPool(const MysqlConnPool&) = delete;
    MysqlConnPool& operator=(const MysqlConnPool&) = delete;

    // 创建单个连接
    DbConn* CreateConn();
    // 定时回收空闲连接线程
    void IdleCheckThread();

    // 数据库配置
    std::string m_host, m_user, m_pwd, m_db;
    int m_port;
    int m_minConn;    // 最小连接数
    int m_maxConn;    // 最大连接数
    int m_idleSec;    // 空闲超时秒
    int m_getTimeout; // 获取连接等待超时

    std::queue<DbConn*> m_idleQueue; // 空闲连接队列
    std::atomic<int> m_totalConn;    // 当前总连接数
    std::mutex m_mtx;
    std::condition_variable m_cv;
    std::thread m_checkThread;
    std::atomic<bool> m_running;      // 池运行标记
};

// RAII自动归还连接智能守卫
class ConnGuard
{
public:
    ConnGuard(DbConn* c, MysqlConnPool& pool) : m_conn(c), m_pool(pool) {}
    ~ConnGuard()
    {
        if (m_conn) m_pool.RetConn(m_conn);
    }
    // 禁止拷贝
    ConnGuard(const ConnGuard&) = delete;
    ConnGuard& operator=(const ConnGuard&) = delete;
    // 移动构造
    ConnGuard(ConnGuard&& other) noexcept
    {
        m_conn = other.m_conn;
        m_pool = other.m_pool;
        other.m_conn = nullptr;
    }
    DbConn* Get() { return m_conn; }
private:
    DbConn* m_conn = nullptr;
    MysqlConnPool& m_pool;
};

#endif
```

```cpp
/**
 * @file DBHelper.cpp
 * @brief MySQL 连接池实现文件
 * 
 * 实现了连接池的核心功能：
 * 1. 单例模式获取实例
 * 2. 连接的创建与销毁
 * 3. 连接的获取与归还
 * 4. 空闲连接的定时回收
 * 5. 连接池的关闭与资源释放
 */
#include "ConnPool.h"

/**
 * @brief 获取连接池单例实例（线程安全的懒汉模式）
 * 使用 C++11 static 局部变量特性实现线程安全的单例
 * @return MysqlConnPool& 连接池引用
 */
MysqlConnPool& MysqlConnPool::GetInstance()
{
    static MysqlConnPool pool;  // C++11 保证线程安全的局部静态变量
    return pool;
}

/**
 * @brief 构造函数
 * 初始化连接数为0，运行状态为false
 */
MysqlConnPool::MysqlConnPool()
    : m_totalConn(0), m_running(false)
{}

/**
 * @brief 析构函数
 * 调用 ClosePool() 确保所有连接被正确释放
 */
MysqlConnPool::~MysqlConnPool()
{
    ClosePool();
}

/**
 * @brief 初始化连接池
 * @param host 数据库主机地址
 * @param user 数据库用户名
 * @param pwd 数据库密码
 * @param db 数据库名称
 * @param port 数据库端口
 * @param minConn 最小连接数
 * @param maxConn 最大连接数
 * @param idleTimeout 空闲连接超时时间（秒）
 * @param getConnTimeout 获取连接超时时间（秒）
 * @return bool 初始化成功返回true，失败返回false
 */
bool MysqlConnPool::Init(const std::string& host, const std::string& user, const std::string& pwd,
                         const std::string& db, int port, int minConn, int maxConn,
                         int idleTimeout, int getConnTimeout)
{
    // 设置数据库连接参数
    m_host = host;
    m_user = user;
    m_pwd = pwd;
    m_db = db;
    m_port = port;
    m_minConn = minConn;
    m_maxConn = maxConn;
    m_idleSec = idleTimeout;
    m_getTimeout = getConnTimeout;

    // 预先创建最小连接数目的连接放入空闲队列
    for (int i = 0; i < m_minConn; i++)
    {
        DbConn* conn = CreateConn();
        if (!conn)
        {
            ClosePool();  // 创建失败，清理已创建的连接
            return false;
        }
        m_idleQueue.push(conn);
        m_totalConn++;  // 原子操作，线程安全
    }

    // 启动空闲连接检查线程
    m_running = true;
    m_checkThread = std::thread(&MysqlConnPool::IdleCheckThread, this);
    std::cout << "连接池初始化完成，最小连接:" << m_minConn << " 最大:" << m_maxConn << std::endl;
    return true;
}

/**
 * @brief 创建单个数据库连接
 * @return DbConn* 新创建的连接指针，失败返回nullptr
 */
DbConn* MysqlConnPool::CreateConn()
{
    DbConn* conn = new DbConn();  // 创建连接封装对象
    // 建立与MySQL服务器的连接
    if (!mysql_real_connect(&conn->mysql, m_host.c_str(), m_user.c_str(),
                            m_pwd.c_str(), m_db.c_str(), m_port, nullptr, 0))
    {
        std::cout << "创建连接失败:" << mysql_error(&conn->mysql) << std::endl;
        delete conn;
        return nullptr;
    }
    // 设置字符集为 utf8mb4（支持完整Unicode，包括emoji）
    mysql_set_character_set(&conn->mysql, "utf8mb4");
    return conn;
}

/**
 * @brief 获取数据库连接（线程安全）
 * 采用条件变量实现超时等待机制
 * @return DbConn* 连接指针，超时返回nullptr
 */
DbConn* MysqlConnPool::GetConn()
{
    std::unique_lock<std::mutex> lock(m_mtx);  // 加锁，unique_lock支持条件变量

    // 等待条件：有空闲连接 或 总连接数未达上限，超时返回
    bool ok = m_cv.wait_for(lock, std::chrono::seconds(m_getTimeout), [this]()
    {
        return !m_idleQueue.empty() || m_totalConn < m_maxConn;
    });

    if (!ok)
    {
        std::cout << "获取连接超时，当前总连接:" << m_totalConn << std::endl;
        return nullptr;
    }

    // 优先从空闲队列获取连接
    if (!m_idleQueue.empty())
    {
        DbConn* conn = m_idleQueue.front();
        m_idleQueue.pop();
        conn->Reset();  // 重置连接状态（检测并自动重连）
        return conn;
    }

    // 无空闲连接但未达上限，新建连接
    DbConn* newConn = CreateConn();
    if (newConn) m_totalConn++;
    return newConn;
}

/**
 * @brief 归还连接到连接池（线程安全）
 * @param conn 要归还的连接指针
 */
void MysqlConnPool::RetConn(DbConn* conn)
{
    if (!conn) return;  // 空指针检查

    std::lock_guard<std::mutex> lock(m_mtx);
    m_idleQueue.push(conn);
    m_cv.notify_one();  // 唤醒一个等待获取连接的线程
}

/**
 * @brief 空闲连接定时检查线程函数
 * 每隔30秒检查一次，清理超时的空闲连接
 */
void MysqlConnPool::IdleCheckThread()
{
    while (m_running)
    {
        std::this_thread::sleep_for(std::chrono::seconds(30));  // 每30秒检查一次

        std::lock_guard<std::mutex> lock(m_mtx);
        auto now = std::chrono::steady_clock::now();

        // 清理多余的空闲连接，保留至少minConn个
        while (m_idleQueue.size() > m_minConn)
        {
            DbConn* front = m_idleQueue.front();
            // 计算空闲时间
            auto diff = std::chrono::duration_cast<std::chrono::seconds>(now - front->lastUse).count();

            if (diff > m_idleSec)
            {
                m_idleQueue.pop();
                delete front;      // 释放超时连接
                m_totalConn--;     // 减少总连接数
            }
            else
            {
                break;  // 队首连接未超时，后续连接也不会超时（队列FIFO）
            }
        }
    }
}

/**
 * @brief 关闭连接池，释放所有资源
 */
void MysqlConnPool::ClosePool()
{
    m_running = false;  // 停止空闲检查线程

    // 等待检查线程结束
    if (m_checkThread.joinable())
        m_checkThread.join();

    std::lock_guard<std::mutex> lock(m_mtx);
    // 释放所有空闲连接
    while (!m_idleQueue.empty())
    {
        DbConn* c = m_idleQueue.front();
        m_idleQueue.pop();
        delete c;
    }
    m_totalConn = 0;
    std::cout << "连接池已关闭，所有连接释放" << std::endl;
}
```

```cpp
/**
 * @file main.cpp
 * @brief MySQL 连接池使用示例
 * 
 * 演示如何使用连接池进行数据库操作，包括：
 * 1. 初始化连接池
 * 2. 多线程并发获取连接
 * 3. 执行查询操作
 * 4. 使用预处理语句防止SQL注入
 * 5. RAII自动归还连接
 */
#include "ConnPool.h"
#include <iostream>
#include <thread>

/**
 * @brief 模拟业务线程函数
 * @param tid 线程ID，用于标识不同的业务线程
 */
void BusinessTask(int tid)
{
    // 获取连接池单例实例
    auto& pool = MysqlConnPool::GetInstance();

    // 使用 RAII 守卫自动管理连接生命周期
    // ConnGuard 析构时自动归还连接，无需手动调用 RetConn
    ConnGuard guard(pool.GetConn(), pool);
    DbConn* conn = guard.Get();

    // 检查连接是否获取成功
    if (!conn)
    {
        std::cout << "线程" << tid << " 获取连接失败\n";
        return;
    }

    // ========== 执行查询操作 ==========
    ResultSet res = conn->Query("SELECT id,name,age FROM user LIMIT 5;");
    std::cout << "线程" << tid << " 查询结果：\n";
    for (auto& row : res)
        std::cout << row[0] << " " << row[1] << " " << row[2] << "\n";

    // ========== 使用预处理语句插入数据（防止SQL注入）==========
    // 构建参数列表
    std::vector<std::string> params = {"测试用户" + std::to_string(tid), "20"};
    // 使用占位符 ?，参数通过 PrepareExec 绑定，避免SQL注入
    conn->PrepareExec("INSERT INTO user(name,age) VALUES(?,?)", params);
}

/**
 * @brief 主函数
 * @return int 程序退出码
 */
int main()
{
    // ========== 初始化连接池 ==========
    // 参数说明：host, user, pwd, db, port, minConn, maxConn, idleTimeout, getConnTimeout
    bool initOk = MysqlConnPool::GetInstance().Init(
        "127.0.0.1",   // 数据库主机地址（本地）
        "root",        // 数据库用户名
        "123456",      // 数据库密码
        "test_db",     // 数据库名称
        3306,           // 数据库端口（MySQL默认3306）
        2,              // 最小连接数
        10,             // 最大连接数
        60,             // 空闲连接超时时间（秒）
        3               // 获取连接超时时间（秒）
    );

    if (!initOk)
    {
        std::cerr << "连接池初始化失败" << std::endl;
        return -1;
    }

    // ========== 模拟多线程并发访问 ==========
    std::vector<std::thread> threads;
    // 启动8个业务线程，模拟后端并发请求
    for (int i = 0; i < 8; i++)
        threads.emplace_back(BusinessTask, i);

    // 等待所有线程完成
    for (auto& t : threads)
        t.join();

    // ========== 关闭连接池 ==========
    MysqlConnPool::GetInstance().ClosePool();
    return 0;
}
```
